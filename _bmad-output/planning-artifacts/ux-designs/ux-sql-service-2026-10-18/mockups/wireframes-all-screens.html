Weekly orders by region

Order count and revenue per region for the last full ISO week. Use for the Monday ops review or when sales asks for regional trends; excludes test accounts.

sami · updated 3 days ago
reportingordersLLM-generatedUsed 14× · last 2 weeks ago
SELECT r.name AS region,
       COUNT(o.id) AS orders,
       SUM(o.total_amount) AS revenue
FROM orders o
JOIN regions r ON r.id = o.region_id
WHERE o.created_at >= date_trunc('week', now()) - interval '7 days'
  AND o.created_at < date_trunc('week', now())
  AND NOT o.is_test
GROUP BY r.name
ORDER BY revenue DESC;

SELECT c.id, c.company_name, MIN(o.created_at) AS first_order,
       SUM(o.total_amount) AS lifetime_revenue
FROM customers c JOIN orders o ON o.customer_id = c.id
WHERE o.status = 'paid'
GROUP BY c.id, c.company_name ORDER BY lifetime_revenue DESC LIMIT 50;
